void print_board() {
    move_num++;
    cout << "\n---=== ROUND " << move_num << " ===---\n";
    short j = 8;
    cout << "12345678"; //upper board numbers
    for (short i = 64; i > -1; i--) {
        cout << figure_chars[i];
        if (i % 8 == 0) {
            if (i != 64) {cout << " ";}
            if (i != 64) {cout << j;}
            if (i != 64) {j--;}
            cout << "\n";
        }
    }
}
